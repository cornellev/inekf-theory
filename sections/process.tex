\section{Process Model}

We define the following continuous time (CT) dynamics:

\begin{equation}
  \begin{gathered}
    \dot{\rotmat{W}{B}} = \rotmat{W}{B}\fv{B}{\bar{\bsomega}}^\wedge \\
    \fv{W}{\dbf{v}}_{\mathcal{B}} = \rotmat{W}{B} \fv{B}{\bar{a}} + \fv{W}{g} \\
  \fv{W}{\dbf{p}}_{\mathcal{B}} = \fv{W}{v}_{\mathcal{B}} \\
  \dbf{b}_g = \mathbf{w}_{bg}, \mathbf{w}_{bg} \sim \mathcal{N}(0,\Sigma_g) \\
  \dbf{b}_a = \mathbf{w}_{ba}, \mathbf{w}_{ba} \sim \mathcal{N}(0,\Sigma_a)
  \end{gathered}
  \label{eq:ct_dynamics}
\end{equation}

where $\fv{B}{\bar{\bsomega}}$ and $\fv{B}{\bar{a}}$ are the bias corrected inputs, which are later derived in \ref{sec:measurement_model}. And we know that these dynamics, when \underline{bias-free}, are \textit{group-affine} - that is, that they yield a log-linear error such that $\dot{\bsxi} = A\bsxi$ (with $\bseta = \exp(\bsxi^\wedge)$) - this is further shown in the appendix. However, this is generally not the case, as shown in (\textcolor{red}{CITE Barreau here}), that the augmentation of biases onto the InEKF state yields an imperfect InEKF, as $\bar{\bsomega}$ and $\bar{\mathbf{a}}$ both depend on $\hbf{b}$.
%TODO: add equations for IMU models.

If we assume that the input measurements $\fv{B}{\bsomega}$ and $\fv{B}{a}$ are constant over some small timestep interval $[t, t + \Delta t]$ (i.e., zero-order hold), then we know that the rotation matrix propagation is defined exactly:
%%TODO: add citation on how this actually works.

\begin{equation}
  \rotmat{W}{B}(t_k+s) = \rotmat{W}{B,\mathnormal{k}} \exp(s[\bsomega]^\wedge) 
  \label{eq:rot_prop}
\end{equation}

This means that we can substitute this into the discretization of both velocity and position, with $s = u \Delta t$, for some timestep $\Delta t$. For instance, with velocity, the integrand yields:

\begin{equation}
  \int_0^{\Delta t}\! \exp(s[\bsomega]^\wedge)ds = \Delta t \int_0^1\! \exp(u[\bsphi]^\wedge)du = \Delta t \Gamma_1(\bsphi)
  \label{eq:vel_integration}
\end{equation}

And the integral behaves similarly for position. This means that after discretizing, we are left with the following dynamics that obey the $SO(3)$ manifold that live on $SE_2(3)$, with $\bsphi \triangleq \bar{\bsomega} \Delta t$, and frames omitted from state components as in \eqref{eq:car_state} for brevity:

\small{
\begin{equation}
  \begin{aligned}
    R_{k+1|k} &= R_{k|k}\Gamma_0(\bsphi) \\
    \mathbf{v}_{k+1|k} &= \mathbf{v}_{k|k} + \fv{W}{g}\Delta t + R_{k|k}\Gamma_1(\bsphi)\mathbf{a}\Delta t \\
    \mathbf{p}_{k+1|k} &= \mathbf{p}_{k|k} + \mathbf{v}_{k|k}\Delta t + \frac{1}{2}\fv{W}{g}\Delta t^2 + R_{k|k}\Gamma_2(\bsphi)\mathbf{a}\Delta t^2 \\
  \end{aligned}
  \label{eq:dt_dynamics}
\end{equation}
}

\subsection{Component-wise Error Derivations}

\subsubsection{Rotation Error}
Using the right-invariant error convention as defined in \eqref{eq:ri_error}, we can decompose the rotation error as:

\begin{equation}
  \begin{gathered}
  \dot{\eta}_R = \dot{\hat{R}}R^\top + \hat{R}\dot{R}^\top = \hat{R}\bar{\bsomega}^\wedge R^\top - \hat{R}\bsomega^\wedge R^\top \\
  \dot{\eta}_R = \hat{R}(\bar{\bsomega}-\bsomega)^\wedge R^\top
\end{gathered}
  \label{eq:rotation_error1}
\end{equation}
Using $\bar \omega -\omega  = -\delta b_g+n_g$ ,$Ru^\wedge R^T=(Ru)^\wedge$ ,and $R\delta b_g = \eta_{b_g}$ gets us 
\begin{equation}
  \dot \eta_R = -\eta_R \eta^\wedge_{b_g}+\eta_R(Rn_g)^\wedge
\end{equation}
Pattern matching with rotation ate of change gives us $\dot \eta_R = -\eta_R\hat \omega_{b}$. We also know that $\omega_b = J_r(\xi_R)\dot \xi_R$.\\
\begin{equation}
  \dot \xi_R = J^{-1}_r(\xi_R)\omega_b=-J_r^{-1}(\xi_R)\eta_{b_g}
\end{equation}
\begin{equation}
  =-J_r^{-1}(\xi_R)J_r(\xi_R)\xi_{b_g}
  )

\end{equation}

\dot \eta_y = \dot R(\hat{\mathbf{y}} - \mathbf{y}) + R(\dot{\hat{\mathbf{y}}} - \dot{\mathbf{y}})

\subsubsection{Velocity Error}
$$\eta_v = \hat v - \eta_r v$$
$$\dot \eta_v = \dot \hat v - \dot \eta_r v - \eta_r \dot v$$
$$
= (\hat R\bar{{a}} + {g}) - \dot\eta_R{v} - \eta_R(R{a} + {g})
$$
$$=\hat R \bar a - \eta_R Ra+g-\dot \eta_Rv - \eta_R g$$
Now, let us write useful identities. 
\begin{equation}
  \begin{aligned}
  \eta_r R = \hat R\\
  \hat R \bar a -\eta_R R a = \hat R(\bar a - a)\\
   = \eta_R R(-\delta b_a + n_a)=\\
   -\eta_r \eta_{b_a}+\hat R n_a
  \end{aligned}

\end{equation}
Using these, we get 
$$\dot \eta_v = -\eta_R \eta_{b_a}+\hat R n_a +(I-\eta_R)g-\dot \eta_R v $$

We now expand the final term. From ??,
$-\dot\eta_R = \eta_R\eta_{b_g}^\wedge - \eta_R(R{n}_g)^\wedge$, so
\begin{equation}
  -\dot\eta_R{v}
    = \eta_R\eta_{b_g}^\wedge{v} - \eta_R(R{n}_g)^\wedge{v}
\end{equation}

Substituting ${v} = \eta_R^\top(\hat{{v}} - \eta_v)$ and applying
the conjugation identity $\eta_R\mathbf{u}^\wedge\eta_R^\top =
(\eta_R\mathbf{u})^\wedge$:
\begin{equation}
  \begin{aligned}
    \eta_R\eta_{b_g}^\wedge\mathbf{v}
      &= \eta_R\eta_{b_g}^\wedge\eta_R^\top(\hat{\mathbf{v}} - \bseta_v) \\
      &= (\eta_R\bseta_{b_g})^\wedge(\hat{\mathbf{v}} - \bseta_v)
  \end{aligned}
\end{equation}

Retaining only first-order terms, $\eta_R \approx I$ and the product
$(\cdot)^\wedge\bseta_v$ is second order, so
\begin{equation}
  \begin{aligned}
    (\eta_R\bseta_{b_g})^\wedge(\hat{\mathbf{v}} - \bseta_v)
      &\approx \bseta_{b_g}^\wedge\hat{\mathbf{v}}
       = -\hat{\mathbf{v}}^\wedge\bsxi_{b_g} \\
    -\eta_R(R\mathbf{n}_g)^\wedge\mathbf{v}
      &\approx -(\hat{R}\mathbf{n}_g)^\wedge\hat{\mathbf{v}}
       = \hat{\mathbf{v}}^\wedge\hat{R}\mathbf{n}_g
  \end{aligned}
\end{equation}

The remaining two terms linearize as
\begin{equation}
  \begin{aligned}
    (I - \eta_R)\mathbf{g} &\approx -\bsxi_R^\wedge\mathbf{g}
      = \mathbf{g}^\wedge\bsxi_R \\
    -\eta_R\bseta_{b_a} &\approx -\bsxi_{b_a}
  \end{aligned}
\end{equation}

Collecting all terms gives the velocity error dynamics:
\begin{equation}
  \dot{\bsxi}_v = \mathbf{g}^\wedge\bsxi_R - \hat{\mathbf{v}}^\wedge\bsxi_{b_g}
    - \bsxi_{b_a} + \hat{\mathbf{v}}^\wedge\hat{R}\mathbf{n}_g
    + \hat{R}\mathbf{n}_a
  \label{eq:vel_error}
\end{equation}
